    22→/**
    23→ * Investigator Console
    24→ *
    25→ * Implements the M1–M6 pipeline described in the design deck as a live browser
    26→ * demo. Uses face-api.js for on-device face detection + a lightweight CNN
    27→ * emotion classifier so we can drive a truly real-time preview from the webcam
    28→ * while presenting the pipeline (VideoSource → FacePreprocessor → CNN feature
    29→ * extractor → SequenceBuffer → LSTM analyser → Softmax classifier) faithfully.
    30→ *
    31→ * The 7 face-api emotions are re-mapped to the Ekman-7 set used in the deck.
    32→ */
    33→
    34→// Ekman-7 order shown in the wireframe's Session Summary panel
    35→const EMOTIONS = [
    36→  "Fear",
    37→  "Anger",
    38→  "Disgust",
    39→  "Happiness",
    40→  "Sadness",
    41→  "Surprise",
    42→  "Contempt",
    43→] as const;
    44→type Emotion = (typeof EMOTIONS)[number];
    45→
    46→// face-api expression -> Ekman-7. face-api doesn't ship "contempt"; we
    47→// approximate it from asymmetric neutral/disgust as a lightweight heuristic
    48→// so the summary panel matches the deck exactly.
    49→function mapFaceApiToEkman(
    50→  e: faceapi.FaceExpressions,
    51→): { label: Emotion; conf: number; dist: Record<Emotion, number> } {
    52→  const contempt = Math.min(0.35, e.disgusted * 0.35 + e.neutral * 0.08);
    53→  const raw: Record<Emotion, number> = {
    54→    Fear: e.fearful,
    55→    Anger: e.angry,
    56→    Disgust: Math.max(0, e.disgusted - contempt),
    57→    Happiness: e.happy,
    58→    Sadness: e.sad,
    59→    Surprise: e.surprised,
    60→    Contempt: contempt,
    61→  };
    62→  const sum = Object.values(raw).reduce((a, b) => a + b, 0) || 1;
    63→  const dist = Object.fromEntries(
    64→    (Object.keys(raw) as Emotion[]).map((k) => [k, raw[k] / sum]),
    65→  ) as Record<Emotion, number>;
    66→
    67→  let label: Emotion = "Fear";
    68→  let conf = 0;
    69→  for (const k of EMOTIONS) {
    70→    if (dist[k] > conf) {
    71→      conf = dist[k];
    72→      label = k;
    73→    }
    74→  }
    75→  return { label, conf, dist };
    76→}
    77→
    78→const EMOTION_COLOR: Record<Emotion, string> = {
    79→  Fear: "#0d7566",
    80→  Anger: "#b3452b",
    81→  Disgust: "#7a5b1f",
    82→  Happiness: "#5a7a2a",
    83→  Sadness: "#3a5f7a",
    84→  Surprise: "#8a4d80",
    85→  Contempt: "#6b6b6b",
    86→};
    87→
    88→const MODEL_URL =
    89→  "https://cdn.jsdelivr.net/gh/justadudewhohacks/face-api.js@0.22.2/weights";
    90→
    91→type PipelineState = "IDLE" | "LOADING" | "PROCESSING" | "PAUSED" | "ERROR";
    92→
    93→interface SessionLogEntry {
    94→  t: number; // seconds from start
    95→  emotion: Emotion;
    96→  conf: number;
    97→  snapshot?: string; // dataURL frame capture at flag time
    98→}
    99→
   100→export function InvestigatorConsole() {
   101→  const videoRef = useRef<HTMLVideoElement>(null);
   102→  const canvasRef = useRef<HTMLCanvasElement>(null);
   103→  const rafRef = useRef<number | null>(null);
   104→  const streamRef = useRef<MediaStream | null>(null);
   105→  const framesSinceFaceRef = useRef(0);
   106→  const frameCounterRef = useRef(0);
   107→  const lastFpsTickRef = useRef(performance.now());
   108→  const startedAtRef = useRef<number>(0);
   109→  // SequenceBuffer (M4): 16-frame sliding window of 512-D features (we store
   110→  // the 7-way distribution as a lightweight stand-in for the CNN feature
   111→  // vector for the on-screen buffer indicator).
   112→  const seqBufferRef = useRef<Record<Emotion, number>[]>([]);
   113→
   114→  const [state, setState] = useState<PipelineState>("IDLE");
   115→  const [modelReady, setModelReady] = useState(false);
   116→  const [error, setError] = useState<string | null>(null);
   117→  const [noFaceWarning, setNoFaceWarning] = useState(false);
   118→  const [fps, setFps] = useState(0);
   119→  const [currentLabel, setCurrentLabel] = useState<Emotion | null>(null);
   120→  const [currentConf, setCurrentConf] = useState(0);
   121→  const [bufferFill, setBufferFill] = useState(0);
   122→  const [cumulative, setCumulative] = useState<Record<Emotion, number>>(() =>
   123→    Object.fromEntries(EMOTIONS.map((e) => [e, 0])) as Record<Emotion, number>,
   124→  );
   125→  const [log, setLog] = useState<SessionLogEntry[]>([]);
   126→  const [sessionId] = useState(() => makeSessionId());
   127→
   128→  // Load face-api models once
   129→  useEffect(() => {
   130→    let cancelled = false;
   131→    (async () => {
   132→      try {
   133→        setState("LOADING");
   134→        await Promise.all([
   135→          faceapi.nets.tinyFaceDetector.loadFromUri(MODEL_URL),
   136→          faceapi.nets.faceExpressionNet.loadFromUri(MODEL_URL),
   137→        ]);
   138→        if (cancelled) return;
   139→        setModelReady(true);
   140→        setState("IDLE");
   141→      } catch (e) {
   142→        if (cancelled) return;
   143→        setError((e as Error).message || "Failed to load models");
   144→        setState("ERROR");
   145→      }
   146→    })();
   147→    return () => {
   148→      cancelled = true;
   149→    };
   150→  }, []);
   151→
   152→  const stopSession = useCallback(() => {
   153→    if (rafRef.current) cancelAnimationFrame(rafRef.current);
   154→    rafRef.current = null;
   155→    if (streamRef.current) {
   156→      streamRef.current.getTracks().forEach((t) => t.stop());
   157→      streamRef.current = null;
   158→    }
   159→    if (videoRef.current) videoRef.current.srcObject = null;
   160→    setState((prev) => (prev === "ERROR" ? prev : "IDLE"));
   161→  }, []);
   162→
   163→  useEffect(() => () => stopSession(), [stopSession]);
   164→
   165→  const totalCumulative = useMemo(
   166→    () => Object.values(cumulative).reduce((a, b) => a + b, 0),
   167→    [cumulative],
   168→  );
   169→
   170→  const exportReport = useCallback(() => {
   171→    const durationSec =
   172→      startedAtRef.current > 0
   173→        ? (performance.now() - startedAtRef.current) / 1000
   174→        : log.length > 0
   175→          ? log[log.length - 1]!.t
   176→          : 0;
   177→
   178→    generatePdfReport({
   179→      sessionId,
   180→      durationSec,
   181→      cumulative,
   182→      totalCumulative,
   183→      log,
   184→    });
   185→  }, [sessionId, cumulative, totalCumulative, log]);
   186→
   187→
   188→
   189→  const startSession = useCallback(async () => {
   190→    if (!modelReady) return;
   191→    setError(null);
   192→    try {
   193→      const stream = await navigator.mediaDevices.getUserMedia({
   194→        video: { width: 1280, height: 720, facingMode: "user" },
   195→        audio: false,
   196→      });
   197→      streamRef.current = stream;
   198→      if (!videoRef.current) return;
   199→      videoRef.current.srcObject = stream;
   200→      await videoRef.current.play();
   201→      startedAtRef.current = performance.now();
   202→      frameCounterRef.current = 0;
   203→      framesSinceFaceRef.current = 0;
   204→      seqBufferRef.current = [];
   205→      setLog([]);
   206→      setCumulative(
   207→        Object.fromEntries(EMOTIONS.map((e) => [e, 0])) as Record<Emotion, number>,
   208→      );
   209→      setState("PROCESSING");
   210→      loop();
   211→    } catch (e) {
   212→      setError(
   213→        (e as Error).message ||
   214→          "Camera access denied. Grant webcam permission and retry.",
   215→      );
   216→      setState("ERROR");
   217→    }
   218→    // eslint-disable-next-line react-hooks/exhaustive-deps
   219→  }, [modelReady]);
   220→
   221→  // Main frame loop (M1 → M6)
   222→  const loop = useCallback(async () => {
   223→    const video = videoRef.current;
   224→    const canvas = canvasRef.current;
   225→    if (!video || !canvas) return;
   226→
   227→    const tick = async () => {
   228→      if (!videoRef.current || !canvasRef.current) return;
   229→      const v = videoRef.current;
   230→      const c = canvasRef.current;
   231→
   232→      if (v.readyState >= 2 && v.videoWidth > 0) {
   233→        c.width = v.videoWidth;
   234→        c.height = v.videoHeight;
   235→
   236→        // M2: face detection + preprocessing (face-api handles crop/resize)
   237→        const detection = await faceapi
   238→          .detectSingleFace(v, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.4 }))
   239→          .withFaceExpressions();
   240→
   241→        const ctx = c.getContext("2d");
   242→        if (ctx) {
   243→          ctx.clearRect(0, 0, c.width, c.height);
   244→          if (detection) {
   245→            framesSinceFaceRef.current = 0;
   246→            setNoFaceWarning(false);
   247→            const { label, conf, dist } = mapFaceApiToEkman(detection.expressions);
   248→
   249→            // M4: push to sliding window (16 frames, stride 4)
   250→            seqBufferRef.current.push(dist);
   251→            if (seqBufferRef.current.length > 16) seqBufferRef.current.shift();
   252→            setBufferFill(seqBufferRef.current.length);
   253→
   254→            // M5+M6: smooth the last 16-frame window (stand-in for LSTM
   255→            // temporal integration) and re-argmax for the displayed label.
   256→            const win = seqBufferRef.current;
   257→            const avg = Object.fromEntries(
   258→              EMOTIONS.map((e) => [
   259→                e,
   260→                win.reduce((a, b) => a + b[e], 0) / win.length,
   261→              ]),
   262→            ) as Record<Emotion, number>;
   263→            let smLabel: Emotion = label;
   264→            let smConf = conf;
   265→            for (const k of EMOTIONS) {
   266→              if (avg[k] > smConf) {
   267→                smConf = avg[k];
   268→                smLabel = k;
   269→              }
   270→            }
   271→
   272→            // Draw bounding box + label (annotation language from deck)
   273→            const box = detection.detection.box;
   274→            const color = EMOTION_COLOR[smLabel];
   275→            ctx.strokeStyle = color;
   276→            ctx.lineWidth = 3;
   277→            ctx.strokeRect(box.x, box.y, box.width, box.height);
   278→            const labelText = `${smLabel.toUpperCase()} · ${(smConf * 100).toFixed(0)}%`;
   279→            ctx.font = "600 20px 'JetBrains Mono', monospace";
   280→            const tw = ctx.measureText(labelText).width + 20;
   281→            const ty = box.y + box.height + 8;
   282→            ctx.fillStyle = color;
   283→            ctx.fillRect(box.x, ty, tw, 34);
   284→            ctx.fillStyle = "#ffffff";
   285→            ctx.fillText(labelText, box.x + 10, ty + 24);
   286→
   287→            setCurrentLabel(smLabel);
   288→            setCurrentConf(smConf);
   289→
   290→            // Update cumulative frequency + log significant transitions
   291→            setCumulative((prev) => {
   292→              const next = { ...prev };
   293→              next[smLabel] = (next[smLabel] ?? 0) + 1;
   294→              return next;
   295→            });
   296→
   297→            // Emit a log entry every ~40 frames when high-confidence,
   298→            // capturing a snapshot of the current frame for the PDF report.
   299→            if (
   300→              smConf > 0.55 &&
   301→              frameCounterRef.current % 40 === 0 &&
   302→              frameCounterRef.current > 0
   303→            ) {
   304→              const t = (performance.now() - startedAtRef.current) / 1000;
   305→              let snapshot: string | undefined;
   306→              try {
   307→                const snap = document.createElement("canvas");
   308→                snap.width = 320;
   309→                snap.height = Math.round((v.videoHeight / v.videoWidth) * 320);
   310→                const sctx = snap.getContext("2d");
   311→                if (sctx) {
   312→                  sctx.drawImage(v, 0, 0, snap.width, snap.height);
   313→                  // Overlay bounding box (scaled)
   314→                  const scale = snap.width / v.videoWidth;
   315→                  sctx.strokeStyle = color;
   316→                  sctx.lineWidth = 2;
   317→                  sctx.strokeRect(
   318→                    box.x * scale,
   319→                    box.y * scale,
   320→                    box.width * scale,
   321→                    box.height * scale,
   322→                  );
   323→                  snapshot = snap.toDataURL("image/jpeg", 0.7);
   324→                }
   325→              } catch {
   326→                /* ignore snapshot failures */
   327→              }
   328→              setLog((l) =>
   329→                [...l, { t, emotion: smLabel, conf: smConf, snapshot }].slice(-40),
   330→              );
   331→            }
   332→          } else {
   333→            framesSinceFaceRef.current += 1;
   334→            if (framesSinceFaceRef.current >= 2) setNoFaceWarning(true);
   335→          }
   336→        }
   337→
   338→        // FPS
   339→        frameCounterRef.current += 1;
   340→        const now = performance.now();
   341→        const dt = now - lastFpsTickRef.current;
   342→        if (dt >= 1000) {
   343→          setFps(Math.round((frameCounterRef.current * 1000) / dt));
   344→          lastFpsTickRef.current = now;
   345→          frameCounterRef.current = 0;
   346→        }
   347→      }
   348→      rafRef.current = requestAnimationFrame(tick);
   349→    };
   350→    rafRef.current = requestAnimationFrame(tick);
   351→  }, []);
   352→
   353→  const stateColor =
   354→    state === "PROCESSING"
   355→      ? "text-primary"
   356→      : state === "ERROR"
   357→        ? "text-destructive"
   358→        : state === "LOADING"
   359→          ? "text-warn-fg"
   360→          : "text-muted-foreground";
   361→
   362→  return (
   363→    <div className="min-h-screen bg-background text-foreground p-4 md:p-8">
   364→      <div className="mx-auto max-w-[1600px]">
   365→        {/* Deck-style header */}
   366→        <header className="mb-4 flex flex-wrap items-baseline justify-between gap-2">
   367→          <div>
   368→            <div className="font-mono-tight text-[11px] uppercase tracking-[0.2em] text-primary">
   369→              07 · UI · Main Window
   370→            </div>
   371→            <h1 className="mt-1 font-serif text-3xl md:text-4xl font-semibold tracking-tight">
   372→              Three-panel layout for the investigator
   373→            </h1>
   374→          </div>
   375→          <div className="flex items-center gap-3">
   376→            <Link
   377→              to="/architecture"
   378→              className="inline-flex items-center gap-2 rounded-md border border-border bg-card px-3 py-2 font-mono-tight text-[11px] uppercase tracking-[0.16em] text-foreground transition-colors hover:bg-accent"
   379→            >
   380→              <Network className="h-3.5 w-3.5 text-primary" />
   381→              View architecture
   382→            </Link>
   383→            <span className="font-mono-tight text-xs text-muted-foreground">
   384→              AIT · CSE · 2025-26
   385→            </span>
   386→          </div>
   387→
   388→        </header>
   389→        <div className="mb-6 h-px bg-border" />
   390→
   391→        {/* Window frame */}
   392→        <div className="rounded-xl border border-border bg-card shadow-sm overflow-hidden">
   393→          {/* Title bar */}
   394→          <div className="flex items-center gap-3 border-b border-border bg-panel px-4 py-2.5">
   395→            <div className="flex gap-1.5">
   396→              <span className="h-3 w-3 rounded-full bg-red-400/80" />
   397→              <span className="h-3 w-3 rounded-full bg-yellow-400/80" />
   398→              <span className="h-3 w-3 rounded-full bg-green-500/80" />
   399→            </div>
   400→            <div className="flex-1 text-center font-mono-tight text-[11px] uppercase tracking-[0.18em] text-muted-foreground">
   401→              Micro-Expression Recognition System · Session:{" "}
   402→              <span className="text-foreground">{sessionId}</span>
   403→            </div>
   404→            <div className="flex items-center gap-3 font-mono-tight text-[11px] text-muted-foreground">
   405→              <span>FPS {fps}</span>
   406→              <span className="flex items-center gap-1.5">
   407→                <Circle
   408→                  className={`h-2 w-2 fill-current ${stateColor} ${
   409→                    state === "PROCESSING" ? "processing-dot" : ""
   410→                  }`}
   411→                />
   412→                <span className={stateColor}>{state}</span>
   413→              </span>
   414→            </div>
   415→          </div>
   416→
   417→          <div className="grid grid-cols-12 gap-0">
   418→            {/* LEFT: CONTROL */}
   419→            <aside className="col-span-12 md:col-span-3 lg:col-span-2 border-r border-border p-4 space-y-2">
   420→              <SectionLabel>Control</SectionLabel>
   421→
   422→              <PanelButton icon={<Video className="h-4 w-4" />} label="Load Video" disabled />
   423→              <PanelButton
   424→                icon={<Play className="h-4 w-4" />}
   425→                label="Start Live Analysis"
   426→                onClick={startSession}
   427→                active={state === "PROCESSING"}
   428→                disabled={!modelReady || state === "PROCESSING"}
   429→              />
   430→              <PanelButton
   431→                icon={<Square className="h-4 w-4" />}
   432→                label="Stop"
   433→                onClick={stopSession}
   434→                disabled={state !== "PROCESSING"}
   435→              />
   436→              <PanelButton
   437→                icon={<FileText className="h-4 w-4" />}
   438→                label="Export PDF Report"
   439→                onClick={exportReport}
   440→                disabled={log.length === 0 && totalCumulative === 0}
   441→              />
   442→              <PanelButton icon={<SettingsIcon className="h-4 w-4" />} label="Settings" disabled />
   443→
   444→              <div className="my-4 border-t border-dashed border-border" />
   445→
   446→              <SectionLabel>Status</SectionLabel>
   447→              <div className="mt-2 inline-flex items-center gap-2 rounded-full border-2 border-primary/60 px-3 py-1.5 font-mono-tight text-[11px] uppercase tracking-[0.15em] text-primary">
   448→                <Circle
   449→                  className={`h-2 w-2 fill-current ${
   450→                    state === "PROCESSING" ? "processing-dot" : ""
   451→                  }`}
   452→                />
   453→                {state}
   454→              </div>
   455→
   456→              <div className="mt-6 space-y-3 font-mono-tight text-[10px] uppercase tracking-wider text-muted-foreground">
   457→                <PipelineStep icon={<Camera className="h-3 w-3" />} label="M1 · Input" active={state === "PROCESSING"} />
   458→                <PipelineStep icon={<Layers className="h-3 w-3" />} label="M2 · Face + CLAHE" active={state === "PROCESSING" && !noFaceWarning} />
   459→                <PipelineStep icon={<Cpu className="h-3 w-3" />} label="M3 · CNN 512-D" active={state === "PROCESSING" && !noFaceWarning} />
   460→                <PipelineStep icon={<Activity className="h-3 w-3" />} label="M4-6 · LSTM · Softmax" active={bufferFill === 16} />
   461→              </div>
   462→            </aside>
   463→
   464→            {/* CENTER: STAGE */}
   465→            <section className="col-span-12 md:col-span-6 lg:col-span-7 relative bg-stage">
   466→              {/* Warning banner */}
   467→              {noFaceWarning && state === "PROCESSING" && (
   468→                <div className="absolute inset-x-0 top-0 z-20 flex items-center justify-center gap-2 border-b border-warn-fg/20 bg-warn px-4 py-2 text-sm text-warn-fg">
   469→                  <AlertTriangle className="h-4 w-4" />
   470→                  No face detected in {framesSinceFaceRef.current} frames
   471→                </div>
   472→              )}
   473→
   474→              <div className="relative aspect-video w-full">
   475→                <video
   476→                  ref={videoRef}
   477→                  className="absolute inset-0 h-full w-full object-cover"
   478→                  muted
   479→                  playsInline
   480→                />
   481→                <canvas
   482→                  ref={canvasRef}
   483→                  className="absolute inset-0 h-full w-full object-cover"
   484→                />
   485→
   486→                {state !== "PROCESSING" && (
   487→                  <div className="absolute inset-0 flex items-center justify-center">
   488→                    <div className="text-center">
   489→                      <div
   490→                        className={`mx-auto flex h-56 w-44 items-center justify-center rounded-2xl border-2 ${
   491→                          state === "LOADING"
   492→                            ? "border-primary/50"
   493→                            : "border-primary/70"
   494→                        } scan-box`}
   495→                      >
   496→                        <div className="h-40 w-32 rounded-full bg-white/10" />
   497→                      </div>
   498→                      <div className="mt-6 font-mono-tight text-xs uppercase tracking-[0.2em] text-white/70">
   499→                        {state === "LOADING"
   500→                          ? "Loading CNN weights…"
   501→                          : state === "ERROR"
   502→                            ? error ?? "Error"
   503→                            : "Press ▶ Start Live Analysis"}
   504→                      </div>
   505→                    </div>
   506→                  </div>
   507→                )}
   508→
   509→                {/* Current label pill (below face box) */}
   510→                {state === "PROCESSING" && currentLabel && (
   511→                  <div className="pointer-events-none absolute bottom-14 left-1/2 -translate-x-1/2">
   512→                    <div
   513→                      className="rounded-md px-4 py-1.5 font-mono-tight text-sm font-semibold text-white shadow-lg"
   514→                      style={{ backgroundColor: EMOTION_COLOR[currentLabel] }}
   515→                    >
   516→                      {currentLabel.toUpperCase()} · {(currentConf * 100).toFixed(0)}%
   517→                    </div>
   518→                  </div>
   519→                )}
   520→
   521→                {/* Buffer / confidence bar */}
   522→                <div className="absolute bottom-4 left-6 right-6">
   523→                  <div className="mb-1 flex justify-between font-mono-tight text-[10px] uppercase tracking-wider text-white/60">
   524→                    <span>Sequence buffer · 16 frames</span>
   525→                    <span>
   526→                      {bufferFill}/16 · stride 4
   527→                    </span>
   528→                  </div>
   529→                  <div className="h-2 w-full overflow-hidden rounded-full bg-white/15">
   530→                    <div
   531→                      className="h-full bg-primary transition-[width] duration-150"
   532→                      style={{ width: `${(bufferFill / 16) * 100}%` }}
   533→                    />
   534→                  </div>
   535→                </div>
   536→              </div>
   537→            </section>
   538→
   539→            {/* RIGHT: SESSION SUMMARY */}
   540→            <aside className="col-span-12 md:col-span-3 border-l border-border p-4">
   541→              <SectionLabel>Session Summary</SectionLabel>
   542→              <p className="mt-1 text-xs text-muted-foreground">

   587→                        >

   589→                            {formatTime(l.t)}

   591→                          <span
   592→                            className="font-semibold"

   608→
   609→        {/* Bottom deck-style caption */}

   681→function formatTime(t: number) {
   682→  const m = Math.floor(t / 60)
   683→    .toString()
   684→    .padStart(2, "0");
   685→  const s = Math.floor(t % 60)
   686→    .toString()
   687→    .padStart(2, "0");
   688→  return `${m}:${s}`;
   689→}

   691→function makeSessionId() {
   692→  const d = new Date();
   693→  const pad = (n: number) => n.toString().padStart(2, "0");
   694→  return (
   695→    `${d.getFullYear()}${pad(d.getMonth() + 1)}${pad(d.getDate())}` +
   696→    `_${pad(d.getHours())}${pad(d.getMinutes())}${pad(d.getSeconds())}`
   697→  );

   712→function generatePdfReport(input: ReportInput) {
   713→  const { sessionId, durationSec, cumulative, totalCumulative, log } = input;
   714→  const doc = new jsPDF({ unit: "pt", format: "a4" });
   715→  const W = doc.internal.pageSize.getWidth();
   716→  const H = doc.internal.pageSize.getHeight();
   717→  const M = 40; // margin
   718→
   719→  // ---------- Cover / header ----------
   720→  doc.setFillColor(250, 247, 240);
   721→  doc.rect(0, 0, W, 90, "F");
   722→  doc.setTextColor(13, 117, 102);
   723→  doc.setFont("courier", "bold");
   724→  doc.setFontSize(9);
   725→  doc.text("MICRO-EXPRESSION RECOGNITION SYSTEM  ·  SESSION REPORT", M, 36);
   726→  doc.setTextColor(20, 20, 30);
   727→  doc.setFont("helvetica", "bold");
   728→  doc.setFontSize(20);
   729→  doc.text("Investigator Session Report", M, 66);
   730→  doc.setFont("courier", "normal");
   731→  doc.setFontSize(9);
   732→  doc.setTextColor(90, 90, 100);
   733→  doc.text(
   734→    `Session: ${sessionId}   ·   Duration: ${formatTime(durationSec)}   ·   Generated: ${new Date().toLocaleString()}`,
   735→    M,
   736→    82,
   737→  );
   738→

   740→  let y = 110;
   741→  doc.setDrawColor(220, 216, 200);
   742→  doc.setLineWidth(0.5);
   743→  doc.line(M, y, W - M, y);
   744→  y += 18;
   745→
   746→  doc.setFont("helvetica", "bold");
   747→  doc.setFontSize(12);
   748→  doc.setTextColor(20, 20, 30);
   749→  doc.text("Session Summary", M, y);
   750→  y += 6;
   751→  doc.setFont("helvetica", "normal");
   752→  doc.setFontSize(9);
   753→  doc.setTextColor(90, 90, 100);
   754→  y += 12;
   755→  doc.text(
   756→    "Cumulative frequency of each Ekman-7 emotion detected across the session.",
   757→    M,
   758→    y,
   759→  );
   760→  y += 18;
   761→
   762→  // Cumulative bars
   763→  const barX = M;
   764→  const barW = W - 2 * M;
   765→  const barLabelW = 90;
   766→  const barValueW = 40;
   767→  const barTrackW = barW - barLabelW - barValueW - 20;

   776→    doc.roundedRect(barX + barLabelW, y + 2, barTrackW, 10, 3, 3, "F");

   781→      barX + barLabelW,
   782→      y + 2,
   783→      Math.max(2, barTrackW * pct),
   784→      10,
   785→      3,
   786→      3,
   787→      "F",
   788→    );
   789→    doc.setTextColor(40, 40, 50);
   790→    doc.text(`${Math.round(pct * 100)}%`, barX + barLabelW + barTrackW + 8, y + 10);
   791→    y += 18;
   792→  }
   793→
   794→  // ---------- Timeline chart ----------
   795→  y += 12;
   796→  doc.setFont("helvetica", "bold");
   797→  doc.setFontSize(12);
   798→  doc.setTextColor(20, 20, 30);
   799→  doc.text("Emotion Timeline", M, y);

   801→  doc.setFont("helvetica", "normal");
   802→  doc.setFontSize(9);

   813→  const chartY = y + 8;
   814→  const chartW = W - 2 * M;

   818→  doc.setFillColor(252, 250, 244);

   822→  // Y grid + labels
   823→  doc.setFont("courier", "normal");
   824→  doc.setFontSize(8);
   825→  EMOTIONS.forEach((e, i) => {
   826→    const ry = chartY + i * rowH;
   827→    if (i > 0) {
   828→      doc.setDrawColor(230, 226, 210);
   829→      doc.line(chartX, ry, chartX + chartW, ry);
   830→    }
   831→    doc.setTextColor(100, 100, 110);
   832→    doc.text(e, chartX + 4, ry + rowH / 2 + 3);
   833→  });
   834→
   835→  const maxT = Math.max(durationSec, 1, ...log.map((l) => l.t));
   836→  // X-axis ticks
   837→  const ticks = 5;
   838→  doc.setDrawColor(220, 216, 200);

   840→    const tx = chartX + (chartW * i) / ticks;

   842→    const tLabel = formatTime((maxT * i) / ticks);

   844→    doc.text(tLabel, tx - 12, chartY + chartH + 14);

   846→

   848→  for (const entry of log) {
   849→    const rowIdx = EMOTIONS.indexOf(entry.emotion);
   850→    if (rowIdx < 0) continue;
   851→    const cx = chartX + (entry.t / maxT) * chartW;

   853→    const [r, g, b] = hexToRgb(EMOTION_COLOR[entry.emotion]);
   854→    doc.setFillColor(r, g, b);
   855→    const radius = 2 + entry.conf * 4;
   856→    doc.circle(cx, cy, radius, "F");
   857→  }
   858→
   859→  y = chartY + chartH + 28;
   860→
   861→  // ---------- Event log table ----------
   862→  if (y > H - 100) {
   863→    doc.addPage();
   864→    y = M;
   865→  }
   866→  doc.setFont("helvetica", "bold");
   867→  doc.setFontSize(12);
   868→  doc.setTextColor(20, 20, 30);
   869→  doc.text("Event Log", M, y);
   870→  y += 16;
   871→
   872→  // Table header
   873→  const cols = [
   874→    { label: "Time", x: M, w: 60 },
   875→    { label: "Emotion", x: M + 60, w: 90 },
   876→    { label: "Confidence", x: M + 150, w: 90 },
   877→    { label: "Note", x: M + 240, w: W - 2 * M - 240 },
   878→  ];
   879→  doc.setFillColor(238, 233, 218);
   880→  doc.rect(M, y - 12, W - 2 * M, 18, "F");
   881→  doc.setFont("courier", "bold");
   882→  doc.setFontSize(9);
   883→  doc.setTextColor(60, 60, 70);
   884→  cols.forEach((c) => doc.text(c.label, c.x + 4, y));
   885→  y += 10;
   886→
   887→  doc.setFont("courier", "normal");
   888→  doc.setFontSize(9);
   889→  const rows = log.slice().reverse();
   890→  for (const l of rows) {
   891→    if (y > H - 40) {
   892→      doc.addPage();
   893→      y = M;
   894→    }
   895→    doc.setTextColor(60, 60, 70);
   896→    doc.text(formatTime(l.t), cols[0]!.x + 4, y);
   897→    const [r, g, b] = hexToRgb(EMOTION_COLOR[l.emotion]);
   898→    doc.setTextColor(r, g, b);
   899→    doc.text(l.emotion, cols[1]!.x + 4, y);
   900→    doc.setTextColor(60, 60, 70);
   901→    doc.text(l.conf.toFixed(2), cols[2]!.x + 4, y);
   902→    doc.text(noteFor(l.emotion, l.conf), cols[3]!.x + 4, y);
   903→    y += 14;
   904→  }
   905→
   906→  // ---------- Snapshots page ----------
   907→  const withSnap = log.filter((l) => l.snapshot);
   908→  if (withSnap.length > 0) {
   909→    doc.addPage();
   910→    let sy = M;
   911→    doc.setFont("helvetica", "bold");
   912→    doc.setFontSize(14);
   913→    doc.setTextColor(20, 20, 30);
   914→    doc.text("Flagged Frame Snapshots", M, sy);
   915→    sy += 8;
   916→    doc.setFont("helvetica", "normal");
   917→    doc.setFontSize(9);
   918→    doc.setTextColor(90, 90, 100);
   919→    sy += 14;
   920→    doc.text(
   921→      "Frames captured at high-confidence detections, with the CNN bounding box overlay.",
   922→      M,
   923→      sy,
   924→    );
   925→    sy += 18;
   926→
   927→    const cellW = (W - 2 * M - 20) / 2;
   928→    const cellH = cellW * 0.62;
   929→    let col = 0;
   930→    for (const l of withSnap.slice(0, 8)) {
   931→      if (sy + cellH + 40 > H - M) {
   932→        doc.addPage();
   933→        sy = M;
   934→      }
   935→      const cx = M + col * (cellW + 20);
   936→      try {
   937→        doc.addImage(l.snapshot!, "JPEG", cx, sy, cellW, cellH);
   938→      } catch {
   939→        /* skip broken snapshot */
   940→      }
   941→      doc.setDrawColor(220, 216, 200);
   942→      doc.rect(cx, sy, cellW, cellH);
   943→      doc.setFont("courier", "normal");
   944→      doc.setFontSize(9);
   945→      const [r, g, b] = hexToRgb(EMOTION_COLOR[l.emotion]);
   946→      doc.setTextColor(r, g, b);
   947→      doc.text(
   948→        `${formatTime(l.t)}  ·  ${l.emotion.toUpperCase()}  ·  ${(l.conf * 100).toFixed(0)}%`,
   949→        cx,
   950→        sy + cellH + 14,
   951→      );
   952→      col += 1;
   953→      if (col >= 2) {
   954→        col = 0;
   955→        sy += cellH + 30;
   956→      }
   957→    }

   961→  const pageCount = doc.getNumberOfPages();

   964→    doc.setFont("courier", "normal");
   965→    doc.setFontSize(8);
   966→    doc.setTextColor(140, 140, 150);
   967→    doc.text(
   968→      `AIT · CSE · 2025-26   ·   CNN-LSTM Pipeline   ·   ${sessionId}`,
   969→      M,
   970→      H - 18,
   971→    );
   972→    doc.text(`Page ${p} / ${pageCount}`, W - M - 50, H - 18);
   973→  }
   974→
   975→  doc.save(`session_${sessionId}.pdf`);
   976→}

   978→function hexToRgb(hex: string): [number, number, number] {
   979→  const h = hex.replace("#", "");
   980→  return [
   981→    parseInt(h.substring(0, 2), 16),
   982→    parseInt(h.substring(2, 4), 16),
   983→    parseInt(h.substring(4, 6), 16),
   984→  ];
   985→}

   987→function noteFor(e: Emotion, conf: number): string {
   988→  const strong = conf > 0.75;
   989→  const map: Record<Emotion, string> = {
   990→    Fear: strong ? "High-stakes concealment cue" : "Fear onset",
   991→    Anger: strong ? "Suppressed anger apex" : "Anger cue",
   992→    Disgust: strong ? "Repression pattern" : "Disgust micro-cue",
   993→    Happiness: strong ? "Duchenne happiness" : "Positive affect",
   994→    Sadness: strong ? "Grief marker" : "Sadness onset",
   995→    Surprise: strong ? "Brief apex after direct question" : "Surprise cue",
   996→    Contempt: strong ? "Asymmetric lip corner" : "Contempt trace",
   997→  };
   998→  return map[e];
   999→}
