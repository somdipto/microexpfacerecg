    22→    22→/**
    23→    23→ * Investigator Console
    24→    24→ *
    25→    25→ * Implements the M1–M6 pipeline described in the design deck as a live browser
    26→    26→ * demo. Uses face-api.js for on-device face detection + a lightweight CNN
    27→    27→ * emotion classifier so we can drive a truly real-time preview from the webcam
    28→    28→ * while presenting the pipeline (VideoSource → FacePreprocessor → CNN feature
    29→    29→ * extractor → SequenceBuffer → LSTM analyser → Softmax classifier) faithfully.
    30→    30→ *
    31→    31→ * The 7 face-api emotions are re-mapped to the Ekman-7 set used in the deck.
    32→    32→ */
    33→    33→
    34→    34→// Ekman-7 order shown in the wireframe's Session Summary panel
    35→    35→const EMOTIONS = [
    36→    36→  "Fear",
    37→    37→  "Anger",
    38→    38→  "Disgust",
    39→    39→  "Happiness",
    40→    40→  "Sadness",
    41→    41→  "Surprise",
    42→    42→  "Contempt",
    43→    43→] as const;
    44→    44→type Emotion = (typeof EMOTIONS)[number];
    45→    45→
    46→    46→// face-api expression -> Ekman-7. face-api doesn't ship "contempt"; we
    47→    47→// approximate it from asymmetric neutral/disgust as a lightweight heuristic
    48→    48→// so the summary panel matches the deck exactly.
    49→    49→function mapFaceApiToEkman(
    50→    50→  e: faceapi.FaceExpressions,
    51→    51→): { label: Emotion; conf: number; dist: Record<Emotion, number> } {
    52→    52→  const contempt = Math.min(0.35, e.disgusted * 0.35 + e.neutral * 0.08);
    53→    53→  const raw: Record<Emotion, number> = {
    54→    54→    Fear: e.fearful,
    55→    55→    Anger: e.angry,
    56→    56→    Disgust: Math.max(0, e.disgusted - contempt),
    57→    57→    Happiness: e.happy,
    58→    58→    Sadness: e.sad,
    59→    59→    Surprise: e.surprised,
    60→    60→    Contempt: contempt,
    61→    61→  };
    62→    62→  const sum = Object.values(raw).reduce((a, b) => a + b, 0) || 1;
    63→    63→  const dist = Object.fromEntries(
    64→    64→    (Object.keys(raw) as Emotion[]).map((k) => [k, raw[k] / sum]),
    65→    65→  ) as Record<Emotion, number>;
    66→    66→
    67→    67→  let label: Emotion = "Fear";
    68→    68→  let conf = 0;
    69→    69→  for (const k of EMOTIONS) {
    70→    70→    if (dist[k] > conf) {
    71→    71→      conf = dist[k];
    72→    72→      label = k;
    73→    73→    }
    74→    74→  }
    75→    75→  return { label, conf, dist };
    76→    76→}
    77→    77→
    78→    78→const EMOTION_COLOR: Record<Emotion, string> = {
    79→    79→  Fear: "#0d7566",
    80→    80→  Anger: "#b3452b",
    81→    81→  Disgust: "#7a5b1f",
    82→    82→  Happiness: "#5a7a2a",
    83→    83→  Sadness: "#3a5f7a",
    84→    84→  Surprise: "#8a4d80",
    85→    85→  Contempt: "#6b6b6b",
    86→    86→};
    87→    87→
    88→    88→const MODEL_URL =
    89→    89→  "https://cdn.jsdelivr.net/gh/justadudewhohacks/face-api.js@0.22.2/weights";
    90→    90→
    91→    91→type PipelineState = "IDLE" | "LOADING" | "PROCESSING" | "PAUSED" | "ERROR";
    92→    92→
    93→    93→interface SessionLogEntry {
    94→    94→  t: number; // seconds from start
    95→    95→  emotion: Emotion;
    96→    96→  conf: number;
    97→    97→  snapshot?: string; // dataURL frame capture at flag time
    98→    98→}

   100→   100→export function InvestigatorConsole() {
   101→   101→  const videoRef = useRef<HTMLVideoElement>(null);
   102→   102→  const canvasRef = useRef<HTMLCanvasElement>(null);
   103→   103→  const rafRef = useRef<number | null>(null);
   104→   104→  const streamRef = useRef<MediaStream | null>(null);
   105→   105→  const framesSinceFaceRef = useRef(0);
   106→   106→  const frameCounterRef = useRef(0);
   107→   107→  const lastFpsTickRef = useRef(performance.now());
   108→   108→  const startedAtRef = useRef<number>(0);
   109→   109→  // SequenceBuffer (M4): 16-frame sliding window of 512-D features (we store
   110→   110→  // the 7-way distribution as a lightweight stand-in for the CNN feature
   111→   111→  // vector for the on-screen buffer indicator).
   112→   112→  const seqBufferRef = useRef<Record<Emotion, number>[]>([]);
   113→   113→
   114→   114→  const [state, setState] = useState<PipelineState>("IDLE");
   115→   115→  const [modelReady, setModelReady] = useState(false);
   116→   116→  const [error, setError] = useState<string | null>(null);
   117→   117→  const [noFaceWarning, setNoFaceWarning] = useState(false);
   118→   118→  const [fps, setFps] = useState(0);
   119→   119→  const [currentLabel, setCurrentLabel] = useState<Emotion | null>(null);
   120→   120→  const [currentConf, setCurrentConf] = useState(0);
   121→   121→  const [bufferFill, setBufferFill] = useState(0);
   122→   122→  const [cumulative, setCumulative] = useState<Record<Emotion, number>>(() =>
   123→   123→    Object.fromEntries(EMOTIONS.map((e) => [e, 0])) as Record<Emotion, number>,
   124→   124→  );
   125→   125→  const [log, setLog] = useState<SessionLogEntry[]>([]);
   126→   126→  const [sessionId] = useState(() => makeSessionId());
   127→   127→
   128→   128→  // Load face-api models once
   129→   129→  useEffect(() => {
   130→   130→    let cancelled = false;
   131→   131→    (async () => {
   132→   132→      try {
   133→   133→        setState("LOADING");
   134→   134→        await Promise.all([
   135→   135→          faceapi.nets.tinyFaceDetector.loadFromUri(MODEL_URL),
   136→   136→          faceapi.nets.faceExpressionNet.loadFromUri(MODEL_URL),
   137→   137→        ]);
   138→   138→        if (cancelled) return;
   139→   139→        setModelReady(true);
   140→   140→        setState("IDLE");
   141→   141→      } catch (e) {
   142→   142→        if (cancelled) return;
   143→   143→        setError((e as Error).message || "Failed to load models");
   144→   144→        setState("ERROR");
   145→   145→      }
   146→   146→    })();
   147→   147→    return () => {
   148→   148→      cancelled = true;
   149→   149→    };
   150→   150→  }, []);
   151→   151→
   152→   152→  const stopSession = useCallback(() => {
   153→   153→    if (rafRef.current) cancelAnimationFrame(rafRef.current);
   154→   154→    rafRef.current = null;
   155→   155→    if (streamRef.current) {
   156→   156→      streamRef.current.getTracks().forEach((t) => t.stop());
   157→   157→      streamRef.current = null;
   158→   158→    }
   159→   159→    if (videoRef.current) videoRef.current.srcObject = null;
   160→   160→    setState((prev) => (prev === "ERROR" ? prev : "IDLE"));
   161→   161→  }, []);
   162→   162→
   163→   163→  useEffect(() => () => stopSession(), [stopSession]);
   164→   164→
   165→   165→  const totalCumulative = useMemo(
   166→   166→    () => Object.values(cumulative).reduce((a, b) => a + b, 0),
   167→   167→    [cumulative],
   168→   168→  );
   169→   169→
   170→   170→  const exportReport = useCallback(() => {
   171→   171→    const durationSec =
   172→   172→      startedAtRef.current > 0
   173→   173→        ? (performance.now() - startedAtRef.current) / 1000
   174→   174→        : log.length > 0
   175→   175→          ? log[log.length - 1]!.t
   176→   176→          : 0;
   177→   177→
   178→   178→    generatePdfReport({
   179→   179→      sessionId,
   180→   180→      durationSec,
   181→   181→      cumulative,
   182→   182→      totalCumulative,
   183→   183→      log,
   184→   184→    });
   185→   185→  }, [sessionId, cumulative, totalCumulative, log]);
   186→   186→
   187→   187→
   188→   188→
   189→   189→  const startSession = useCallback(async () => {
   190→   190→    if (!modelReady) return;
   191→   191→    setError(null);
   192→   192→    try {
   193→   193→      const stream = await navigator.mediaDevices.getUserMedia({
   194→   194→        video: { width: 1280, height: 720, facingMode: "user" },
   195→   195→        audio: false,
   196→   196→      });
   197→   197→      streamRef.current = stream;
   198→   198→      if (!videoRef.current) return;
   199→   199→      videoRef.current.srcObject = stream;
   200→   200→      await videoRef.current.play();
   201→   201→      startedAtRef.current = performance.now();
   202→   202→      frameCounterRef.current = 0;
   203→   203→      framesSinceFaceRef.current = 0;
   204→   204→      seqBufferRef.current = [];
   205→   205→      setLog([]);
   206→   206→      setCumulative(
   207→   207→        Object.fromEntries(EMOTIONS.map((e) => [e, 0])) as Record<Emotion, number>,
   208→   208→      );
   209→   209→      setState("PROCESSING");
   210→   210→      loop();
   211→   211→    } catch (e) {
   212→   212→      setError(
   213→   213→        (e as Error).message ||
   214→   214→          "Camera access denied. Grant webcam permission and retry.",
   215→   215→      );
   216→   216→      setState("ERROR");
   217→   217→    }
   218→   218→    // eslint-disable-next-line react-hooks/exhaustive-deps
   219→   219→  }, [modelReady]);
   220→   220→
   221→   221→  // Main frame loop (M1 → M6)
   222→   222→  const loop = useCallback(async () => {
   223→   223→    const video = videoRef.current;
   224→   224→    const canvas = canvasRef.current;
   225→   225→    if (!video || !canvas) return;
   226→   226→
   227→   227→    const tick = async () => {
   228→   228→      if (!videoRef.current || !canvasRef.current) return;
   229→   229→      const v = videoRef.current;
   230→   230→      const c = canvasRef.current;
   231→   231→
   232→   232→      if (v.readyState >= 2 && v.videoWidth > 0) {
   233→   233→        c.width = v.videoWidth;
   234→   234→        c.height = v.videoHeight;
   235→   235→
   236→   236→        // M2: face detection + preprocessing (face-api handles crop/resize)
   237→   237→        const detection = await faceapi
   238→   238→          .detectSingleFace(v, new faceapi.TinyFaceDetectorOptions({ inputSize: 224, scoreThreshold: 0.4 }))
   239→   239→          .withFaceExpressions();
   240→   240→
   241→   241→        const ctx = c.getContext("2d");
   242→   242→        if (ctx) {
   243→   243→          ctx.clearRect(0, 0, c.width, c.height);
   244→   244→          if (detection) {
   245→   245→            framesSinceFaceRef.current = 0;
   246→   246→            setNoFaceWarning(false);
   247→   247→            const { label, conf, dist } = mapFaceApiToEkman(detection.expressions);
   248→   248→
   249→   249→            // M4: push to sliding window (16 frames, stride 4)
   250→   250→            seqBufferRef.current.push(dist);
   251→   251→            if (seqBufferRef.current.length > 16) seqBufferRef.current.shift();
   252→   252→            setBufferFill(seqBufferRef.current.length);
   253→   253→
   254→   254→            // M5+M6: smooth the last 16-frame window (stand-in for LSTM
   255→   255→            // temporal integration) and re-argmax for the displayed label.
   256→   256→            const win = seqBufferRef.current;
   257→   257→            const avg = Object.fromEntries(
   258→   258→              EMOTIONS.map((e) => [
   259→   259→                e,
   260→   260→                win.reduce((a, b) => a + b[e], 0) / win.length,
   261→   261→              ]),
   262→   262→            ) as Record<Emotion, number>;
   263→   263→            let smLabel: Emotion = label;
   264→   264→            let smConf = conf;
   265→   265→            for (const k of EMOTIONS) {
   266→   266→              if (avg[k] > smConf) {
   267→   267→                smConf = avg[k];
   268→   268→                smLabel = k;
   269→   269→              }
   270→   270→            }
   271→   271→
   272→   272→            // Draw bounding box + label (annotation language from deck)
   273→   273→            const box = detection.detection.box;
   274→   274→            const color = EMOTION_COLOR[smLabel];
   275→   275→            ctx.strokeStyle = color;
   276→   276→            ctx.lineWidth = 3;
   277→   277→            ctx.strokeRect(box.x, box.y, box.width, box.height);
   278→   278→            const labelText = `${smLabel.toUpperCase()} · ${(smConf * 100).toFixed(0)}%`;
   279→   279→            ctx.font = "600 20px 'JetBrains Mono', monospace";
   280→   280→            const tw = ctx.measureText(labelText).width + 20;
   281→   281→            const ty = box.y + box.height + 8;
   282→   282→            ctx.fillStyle = color;
   283→   283→            ctx.fillRect(box.x, ty, tw, 34);
   284→   284→            ctx.fillStyle = "#ffffff";
   285→   285→            ctx.fillText(labelText, box.x + 10, ty + 24);
   286→   286→
   287→   287→            setCurrentLabel(smLabel);
   288→   288→            setCurrentConf(smConf);
   289→   289→
   290→   290→            // Update cumulative frequency + log significant transitions
   291→   291→            setCumulative((prev) => {
   292→   292→              const next = { ...prev };
   293→   293→              next[smLabel] = (next[smLabel] ?? 0) + 1;
   294→   294→              return next;
   295→   295→            });
   296→   296→
   297→   297→            // Emit a log entry every ~40 frames when high-confidence,
   298→   298→            // capturing a snapshot of the current frame for the PDF report.
   299→   299→            if (
   300→   300→              smConf > 0.55 &&
   301→   301→              frameCounterRef.current % 40 === 0 &&
   302→   302→              frameCounterRef.current > 0
   303→   303→            ) {
   304→   304→              const t = (performance.now() - startedAtRef.current) / 1000;
   305→   305→              let snapshot: string | undefined;
   306→   306→              try {
   307→   307→                const snap = document.createElement("canvas");
   308→   308→                snap.width = 320;
   309→   309→                snap.height = Math.round((v.videoHeight / v.videoWidth) * 320);
   310→   310→                const sctx = snap.getContext("2d");
   311→   311→                if (sctx) {
   312→   312→                  sctx.drawImage(v, 0, 0, snap.width, snap.height);
   313→   313→                  // Overlay bounding box (scaled)
   314→   314→                  const scale = snap.width / v.videoWidth;
   315→   315→                  sctx.strokeStyle = color;
   316→   316→                  sctx.lineWidth = 2;
   317→   317→                  sctx.strokeRect(
   318→   318→                    box.x * scale,
   319→   319→                    box.y * scale,
   320→   320→                    box.width * scale,
   321→   321→                    box.height * scale,
   322→   322→                  );
   323→   323→                  snapshot = snap.toDataURL("image/jpeg", 0.7);
   324→   324→                }
   325→   325→              } catch {
   326→   326→                /* ignore snapshot failures */
   327→   327→              }
   328→   328→              setLog((l) =>
   329→   329→                [...l, { t, emotion: smLabel, conf: smConf, snapshot }].slice(-40),
   330→   330→              );
   331→   331→            }
   332→   332→          } else {
   333→   333→            framesSinceFaceRef.current += 1;
   334→   334→            if (framesSinceFaceRef.current >= 2) setNoFaceWarning(true);
   335→   335→          }
   336→   336→        }
   337→   337→
   338→   338→        // FPS
   339→   339→        frameCounterRef.current += 1;
   340→   340→        const now = performance.now();
   341→   341→        const dt = now - lastFpsTickRef.current;
   342→   342→        if (dt >= 1000) {
   343→   343→          setFps(Math.round((frameCounterRef.current * 1000) / dt));
   344→   344→          lastFpsTickRef.current = now;
   345→   345→          frameCounterRef.current = 0;
   346→   346→        }
   347→   347→      }
   348→   348→      rafRef.current = requestAnimationFrame(tick);
   349→   349→    };
   350→   350→    rafRef.current = requestAnimationFrame(tick);
   351→   351→  }, []);
   352→   352→
   353→   353→  const stateColor =
   354→   354→    state === "PROCESSING"
   355→   355→      ? "text-primary"
   356→   356→      : state === "ERROR"
   357→   357→        ? "text-destructive"
   358→   358→        : state === "LOADING"
   359→   359→          ? "text-warn-fg"
   360→   360→          : "text-muted-foreground";
   361→   361→
   362→   362→  return (
   363→   363→    <div className="min-h-screen bg-background text-foreground p-4 md:p-8">
   364→   364→      <div className="mx-auto max-w-[1600px]">
   365→   365→        {/* Deck-style header */}
   366→   366→        <header className="mb-4 flex flex-wrap items-baseline justify-between gap-2">
   367→   367→          <div>
   368→   368→            <div className="font-mono-tight text-[11px] uppercase tracking-[0.2em] text-primary">
   369→   369→              07 · UI · Main Window
   370→   370→            </div>
   371→   371→            <h1 className="mt-1 font-serif text-3xl md:text-4xl font-semibold tracking-tight">
   372→   372→              Three-panel layout for the investigator
   373→   373→            </h1>
   374→   374→          </div>
   375→   375→          <div className="flex items-center gap-3">
   376→   376→            <Link
   377→   377→              to="/architecture"
   378→   378→              className="inline-flex items-center gap-2 rounded-md border border-border bg-card px-3 py-2 font-mono-tight text-[11px] uppercase tracking-[0.16em] text-foreground transition-colors hover:bg-accent"
   379→   379→            >
   380→   380→              <Network className="h-3.5 w-3.5 text-primary" />
   381→   381→              View architecture
   382→   382→            </Link>
   383→   383→            <span className="font-mono-tight text-xs text-muted-foreground">
   384→   384→              AIT · CSE · 2025-26
   385→   385→            </span>
   386→   386→          </div>
   387→   387→
   388→   388→        </header>
   389→   389→        <div className="mb-6 h-px bg-border" />
   390→   390→
   391→   391→        {/* Window frame */}
   392→   392→        <div className="rounded-xl border border-border bg-card shadow-sm overflow-hidden">
   393→   393→          {/* Title bar */}
   394→   394→          <div className="flex items-center gap-3 border-b border-border bg-panel px-4 py-2.5">
   395→   395→            <div className="flex gap-1.5">
   396→   396→              <span className="h-3 w-3 rounded-full bg-red-400/80" />
   397→   397→              <span className="h-3 w-3 rounded-full bg-yellow-400/80" />
   398→   398→              <span className="h-3 w-3 rounded-full bg-green-500/80" />
   399→   399→            </div>
   400→   400→            <div className="flex-1 text-center font-mono-tight text-[11px] uppercase tracking-[0.18em] text-muted-foreground">
   401→   401→              Micro-Expression Recognition System · Session:{" "}
   402→   402→              <span className="text-foreground">{sessionId}</span>
   403→   403→            </div>
   404→   404→            <div className="flex items-center gap-3 font-mono-tight text-[11px] text-muted-foreground">
   405→   405→              <span>FPS {fps}</span>
   406→   406→              <span className="flex items-center gap-1.5">
   407→   407→                <Circle
   408→   408→                  className={`h-2 w-2 fill-current ${stateColor} ${
   409→   409→                    state === "PROCESSING" ? "processing-dot" : ""
   410→   410→                  }`}
   411→   411→                />
   412→   412→                <span className={stateColor}>{state}</span>
   413→   413→              </span>
   414→   414→            </div>
   415→   415→          </div>
   416→   416→
   417→   417→          <div className="grid grid-cols-12 gap-0">
   418→   418→            {/* LEFT: CONTROL */}
   419→   419→            <aside className="col-span-12 md:col-span-3 lg:col-span-2 border-r border-border p-4 space-y-2">
   420→   420→              <SectionLabel>Control</SectionLabel>
   421→   421→
   422→   422→              <PanelButton icon={<Video className="h-4 w-4" />} label="Load Video" disabled />
   423→   423→              <PanelButton
   424→   424→                icon={<Play className="h-4 w-4" />}
   425→   425→                label="Start Live Analysis"
   426→   426→                onClick={startSession}
   427→   427→                active={state === "PROCESSING"}
   428→   428→                disabled={!modelReady || state === "PROCESSING"}
   429→   429→              />

   444→   444→              <div className="my-4 border-t border-dashed border-border" />
   445→   445→
   446→   446→              <SectionLabel>Status</SectionLabel>
   447→   447→              <div className="mt-2 inline-flex items-center gap-2 rounded-full border-2 border-primary/60 px-3 py-1.5 font-mono-tight text-[11px] uppercase tracking-[0.15em] text-primary">
   448→   448→                <Circle
   449→   449→                  className={`h-2 w-2 fill-current ${
   450→   450→                    state === "PROCESSING" ? "processing-dot" : ""
   451→   451→                  }`}
   452→   452→                />
   453→   453→                {state}
   454→   454→              </div>
   455→   455→
